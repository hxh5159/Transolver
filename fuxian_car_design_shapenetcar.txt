for fold in 0 1 2 3 4 5 6 7 8; do
  python main.py \
    --cfd_model=Transolver \

Step 3：取 9 折平均
对每个指标，把 9 个 evaluation_metrics.json 里的对应键求平均即可。可用的汇总命令（不落盘，只打印）：


python - <<'PY'
import json, glob, numpy as np
files = sorted(glob.glob('output/fold_*/evaluation/evaluation_metrics.json'))
keys = ['volume_velocity_relative_l2_mean', 'surface_pressure_relative_l2_mean']
for k in keys:
    print(k, np.mean([json.load(open(f))[k] for f in files]))
for k in ['mape', 'spearman']:
    print('drag_coefficient.'+k, np.mean([json.load(open(f))['drag_coefficient'][k] for f in files]))
PY
